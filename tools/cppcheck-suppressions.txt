# cppcheck suppressions for spicule, used by tools/lint.sh (`make lint`).
#
# Only categorical false positives belong here -- things cppcheck cannot
# model about a freestanding libc built with -nostdinc on top of ntdll.
# Findings that are merely inconvenient do not; they get reported and
# judged like any other.  Note that lint.sh runs cppcheck with
# --enable=warning,portability only, so the "prefer the _s variant" style
# advice aimed at *callers* of the C library never fires in the first
# place and needs no suppression here.
#
# Deliberately kept short and id-scoped rather than path-scoped: a
# `*:src/string/*.c` style blanket entry would silence real findings along
# with the noise, which defeats the point of running the tool.

# cppcheck cannot see the system headers we deliberately do not include
# (-nostdinc) and reports every one of them as missing.
missingInclude
missingIncludeSystem

# Inline assembly writes through pointer operands with a "memory" clobber.
# cppcheck's value-flow engine has no model for that and reports the
# results as uninitialised reads.
uninitvar:src/math/x87.h
uninitvar:src/math/aarch64_math.h
uninitvar:arch/i386/src/*.c
uninitvar:arch/x86_64/src/*.c
uninitvar:src/setjmp/*

# Reported when a suppression above happens not to match on some arch or
# cppcheck version; not itself a finding about the code.
unmatchedSuppression

# We build with -nostdinc against our own <stdarg.h>, which defines
# va_start/va_arg/va_end as the __builtin_* forms the compilers actually
# implement.  cppcheck's va_list checker only recognises the names it
# expects from a system stdarg.h, so after preprocessing it sees a va_list
# that is never started and reports every use of it.  Applies to every
# variadic function in the tree, not to any particular one.
va_list_usedBeforeStarted

# Ubuntu 24.04's cppcheck 2.13 unwraps uintmax_t to unsigned long long,
# then paradoxically rejects that exact underlying type for %ju.  Every
# affected argument is explicitly uintmax_t (or returned as uintmax_t),
# so these are checker-version false positives; newer cppcheck accepts
# the same calls.  Scope the suppression to the four audited utilities
# rather than disabling printf type checking tree-wide.
invalidPrintfArgType_uint:src/util/cksum.c
invalidPrintfArgType_uint:src/util/dd.c
invalidPrintfArgType_uint:src/util/df.c
invalidPrintfArgType_uint:src/util/du.c

# printf.c reverses a digit buffer into a local (`rev`, `b2`) and emits
# exactly as many bytes as it wrote.  The digit count comes from a
# do/while, so it is >= 1, but cppcheck's value flow drops that lower
# bound, assumes the copy loop runs zero times, and calls the local
# uninitialised.  The length passed to out() is the same count, so a
# zero-length case would read nothing either way.
uninitvar:src/stdio/printf.c

# SIG_DFL/SIG_IGN/SIG_ERR are (void (*)(int))0/1/-1 by definition -- the
# integer-literal-to-pointer casts C requires of <signal.h>, not a
# portability hazard we can write our way out of.
intToPointerCast:src/signal/signal.c
# system() passes SIG_IGN to signal() while it runs the child, same
# integer-literal-to-pointer cast as above.
intToPointerCast:src/stdlib/system.c
# tee(1p) -i passes SIG_IGN to signal() to ignore SIGINT for the
# duration of the copy (XCU tee(1p): "Ignore the SIGINT signal"), same
# integer-literal-to-pointer cast as above.
intToPointerCast:src/util/tee.c
# __plat_sig_sync_kernel() (the Linux rt_sigaction backend) assigns
# SIG_IGN/SIG_DFL to the raw kernel_sigaction's k_handler, same
# integer-literal-to-pointer cast as above -- this is signal()'s own
# storage, one layer lower.
intToPointerCast:src/signal/linux/plat_signal.c

# CLIENT_ID.UniqueProcess/UniqueThread are declared HANDLE by NT but hold
# small integer ids, which is what a pid_t is; the cast back is the
# documented way to read them.  Nothing is truncated: pid_t and the id
# space are both 32-bit on either win32 target.
CastAddressToIntegerAtReturn:src/unistd/nt/plat_unistd.c
# Linux process handles deliberately box a pid in the otherwise opaque
# pointer-shaped platform handle type.  The backend banner documents why
# pid 0 is the null sentinel and why the conversion cannot truncate.
CastAddressToIntegerAtReturn:src/process/linux/plat_process.c

# memmove's overlap test is musl's: the unsigned wrap in `- 2*n` is the
# point of it, computing the "distance is at least n in either direction"
# predicate in one comparison.
signConversion:src/string/memmove.c

# `(int)hi >> 31` in __ashrdi3 is an arithmetic right shift of a 32-bit
# signed value by 31, which is implementation-defined (C99 6.5.7p5), not
# undefined -- only a count >= the width would be.  cppcheck applies its
# left-shift rule to `>>` as well and misreports it; verified against
# cppcheck 2.21.1 with a two-line test case.
shiftTooManyBitsSigned:arch/*/src/int64.c
# arith.c's ev_rel_bit()/wordexp arithmetic `>>` arm ('R' case) already
# validates `0 <= rhs < LONG_BIT` before shifting a signed long by rhs,
# so rhs at LONG_BIT-1 (63 on this LP64 aarch64 target, where LONG_BIT
# is 64) is an in-range width-1 shift, not out-of-range; a negative left
# operand there is implementation-defined (C99 6.5.7p5), not undefined.
# Same misapplication of the left-shift overflow rule to `>>` as above.
shiftTooManyBitsSigned:src/wordexp/arith.c

# cppcheck's bundled posix.cfg models fetestexcept()/feclearexcept()/
# feraiseexcept() after glibc's <fenv.h>: a 6-bit FE_ALL_EXCEPT (glibc's
# __FE_DENORM is bit 1) and an argument range annotation that does not
# match FE_ALL_EXCEPT==0x3d (61) at all ("valid values are '1:31'" --
# stale/wrong even for glibc, which allows 0x3f).  include/fenv.h is a
# different, C99-conformant implementation for this project's own
# arches (see its header comment): a 5-bit FE_ALL_EXCEPT with no
# FE_DENORM, since FE_DENORM is not part of the C99 exception set this
# project claims. Calling any of these three with FE_ALL_EXCEPT is
# exactly correct for *our* <fenv.h>, not a bug. The same stale range
# annotation also misfires on a lone FE_INEXACT (0x20 == 32, still
# outside the bogus '1:31' the annotation allows) -- nearbyint.c's
# snapshot-then-clear-just-FE_INEXACT pattern is the one other caller
# in this tree that passes a single-bit argument outside FE_ALL_EXCEPT.
invalidFunctionArg:src/math/fenv.c
invalidFunctionArg:src/math/nearbyint.c

# fpclassify(x)/signbit(x) (include/math.h) select the f/""/l variant with
# `sizeof(x) == sizeof(float) ? ... : ...` -- the standard pre-_Generic
# type-dispatch idiom this C99 (-std=c99, tcc-compatible, no _Generic)
# tree relies on throughout <math.h>. sizeof of a non-VLA expression never
# evaluates that expression (C99 6.5.3.4p2), so x is not actually called
# here whatever it is syntactically. cproj() and cacosh() are the only two
# call sites in the tree where x itself is a function call (creal(z)/
# cimag(z)) rather than a plain variable, which is what makes cppcheck's
# textual "function call inside sizeof()" check fire.
sizeofFunctionCall:src/complex/complex_parts.c
sizeofFunctionCall:src/complex/cinverse.c

# mmap()'s MAP_FIXED file-backed-remap arm frees m->live/m->locked, then
# calls init_page_state(m, npages), which sets both fields back to NULL
# through the same pointer -- and only then, on a later failure path
# (mlock() below it), frees them again. The second free() is therefore
# free(NULL), a no-op, never the same live pointer twice. cppcheck's
# doubleFree does not track a struct field being reset to NULL inside a
# called function, so it treats the pointer as still live-and-freed
# across that call; reproduced in isolation with an 11-line repro (two
# free()s, an intervening function that clears both fields, and a third
# free() on a conditional path) against cppcheck 2.21.1.
doubleFree:src/mman/mman.c

# ev_shift()'s `>>` arm already validates `0 <= r < 32` (the `if (r < 0 ||
# r >= 32) { ev_fail(e); v = 0; }` immediately above) before shifting a
# signed int32_t by r, so r=31 is the width-1 boundary of an in-range
# shift, not an out-of-range one -- and even a negative left operand at
# that count is implementation-defined by C99 6.5.7p5, not undefined.
# Same misapplication of the left-shift overflow rule to `>>` already
# recorded for arch/*/src/int64.c's `(int)hi >> 31` above, verified
# against cppcheck 2.21.1 there with a two-line test case.
shiftTooManyBitsSigned:src/util/m4.c

# mailx(1) -i ignores SIGINT for the duration of the session (XCU
# mailx(1p): "the SIGINT signal ... shall be ignored"), same
# integer-literal-to-pointer cast as the three call sites already
# suppressed above.
intToPointerCast:src/util/mailx.c

# sort's final write path closes an -o FILE* it fopen()'d, or fflushes
# stdout when writing there instead, in one ternary:
# `outfile ? fclose(outf) != 0 : fflush(outf) != 0`. cppcheck's
# useClosedFile/deallocuse treats the untaken fclose() arm as having run
# before evaluating the fflush() arm, so it reports outf as already
# closed even when outfile is NULL and fclose() never executes.
# Reproduced in an 11-line isolated repro (the same ternary, an FILE*
# initialised from stdout rather than fopen()) against cppcheck 2.21.1.
useClosedFile:src/util/sort.c

# wake_waiters()'s writer-wake loop: the guard immediately above it
# (`if (data->readers || !data->waiting_writers) return;`) only lets
# execution reach the loop when waiting_writers > 0. waiting_writers is
# incremented exactly once, when a write-waiter is linked into the queue,
# and decremented only by unlink_waiter() when a write-waiter is
# unlinked -- the counter and the list contents are kept in lockstep, so
# waiting_writers > 0 guarantees at least one `waiter->write` node is
# still reachable from data->head. The loop therefore always sets `best`
# to a non-NULL value before `data->writer = best->owner`,
# `unlink_waiter(best)`, and `__plat_semaphore_post(best->semaphore)`
# dereference it below. cppcheck's flow-insensitive analysis cannot
# correlate the counter with the list contents, so it assumes the loop
# can run zero iterations and leave best == 0. Reproduced against
# cppcheck 2.21.1 via `tools/lint.sh cppcheck`: 4 nullPointer findings,
# all rooted at the `best = 0` initialisation -- at best->owner,
# unlink_waiter(best), best->semaphore, and the waiter->lock/waiter->linked
# dereferences inside unlink_waiter() itself once called with best.
nullPointer:src/thread/pthread_rwlock.c
